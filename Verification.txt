  ∃ R,

      ∀ (r : ℝ),

            (MeasureTheory.volume (extendedBody A.realization r)).toReal =

def FinitePlanarComplex.ConvexOffsetFormulaAt : FinitePlanarComplex → ℝ → Prop :=
fun C r => (MeasureTheory.volume (offset C.realization r)).toReal = C.steinerPolynomial r
def FinitePlanarComplex.TriangleOffsetFormula : Finset PLPoint → Prop :=
fun s =>
  ∀ (r : ℝ),
    0 ≤ r →
      (MeasureTheory.volume (offset ((convexHull ℝ) ↑s) r)).toReal =
        FinitePlanarComplex.triangleArea s + r * FinitePlanarComplex.trianglePerimeter s + Real.pi * r ^ 2
